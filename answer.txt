import java.util.*;
import java.util.function.*;
import java.util.stream.*;

interface PaperDemand {
    int getRequiredSheets();
}

abstract class PrintJob implements PaperDemand {
    private final int id, ownerId, pages, copies, priority;
    private boolean printed = false;

    public PrintJob(int id, int ownerId, int pages, int copies, int priority) {
        if (id <= 0 || ownerId <= 0) throw new IllegalArgumentException("id/ownerId > 0");
        if (pages < 1 || pages > 100) throw new IllegalArgumentException("pages: 1..100");
        if (copies < 1 || copies > 100) throw new IllegalArgumentException("copies: 1..100");
        if (priority < 1 || priority > 5) throw new IllegalArgumentException("priority: 1..5");
        this.id = id; this.ownerId = ownerId; this.pages = pages;
        this.copies = copies; this.priority = priority;
    }

    public int getId() { return id; }
    public int getOwnerId() { return ownerId; }
    public int getPages() { return pages; }
    public int getCopies() { return copies; }
    public int getPriority() { return priority; }
    public boolean isPrinted() { return printed; }
    public void markPrinted() { printed = true; }
    public abstract String getDetails();
    protected String statusText() { return printed ? "напечатано" : "ожидает"; }
}

class SingleSidedJob extends PrintJob {
    public SingleSidedJob(int id, int ownerId, int pages, int copies, int priority) {
        super(id, ownerId, pages, copies, priority);
    }
    @Override public int getRequiredSheets() { return getPages() * getCopies(); }
    @Override public String getDetails() {
        return "SingleSidedJob{id=" + getId() + ", владелец=" + getOwnerId()
                + ", листов=" + getRequiredSheets() + ", статус=" + statusText() + "}";
    }
}

class DoubleSidedJob extends PrintJob {
    public DoubleSidedJob(int id, int ownerId, int pages, int copies, int priority) {
        super(id, ownerId, pages, copies, priority);
    }
    @Override public int getRequiredSheets() { return ((getPages() + 1) / 2) * getCopies(); }
    @Override public String getDetails() {
        return "DoubleSidedJob{id=" + getId() + ", владелец=" + getOwnerId()
                + ", листов=" + getRequiredSheets() + ", статус=" + statusText() + "}";
    }
}

class PrintQueue<T extends PrintJob> {
    private static final int MAX_PAPER = 1_000_000;
    private final List<T> jobs = new ArrayList<>();
    private int paper;

    public PrintQueue(int paper) {
        if (paper < 0 || paper > MAX_PAPER) throw new IllegalArgumentException("paper: 0..1000000");
        this.paper = paper;
    }

    public boolean add(T job) {
        if (job == null || job.isPrinted()) return false;
        if (jobs.stream().anyMatch(j -> j.getId() == job.getId())) return false;
        jobs.add(job);
        return true;
    }

    public int getPaper() { return paper; }

    public boolean restock(int amount) {
        if (amount <= 0 || (long) paper + amount > MAX_PAPER) return false;
        paper += amount;
        return true;
    }

    public Stream<T> getAnalyticsStream() { return new ArrayList<>(jobs).stream(); }

    public List<T> process(Predicate<T> condition, Comparator<T> order, Consumer<T> action) {
        List<T> pending = jobs.stream().filter(j -> !j.isPrinted()).sorted(order)
                .collect(Collectors.toList());
        List<T> done = new ArrayList<>();
        for (T job : pending) {
            int need = job.getRequiredSheets();
            if (need <= paper && condition.test(job)) {
                paper -= need;
                action.accept(job);
                done.add(job);
            }
        }
        return done;
    }
}

    private static List<PrintJob> processB4() {
        final int reserve = (queue.getPaper() + 3) / 4;   // считается один раз
        Predicate<PrintJob> condition = j -> queue.getPaper() - j.getRequiredSheets() >= reserve;
        Comparator<PrintJob> order = Comparator.comparingInt(PrintJob::getPriority).reversed()
                .thenComparingInt(PrintJob::getRequiredSheets)
                .thenComparingInt(PrintJob::getId);
        Consumer<PrintJob> action = PrintJob::markPrinted;
        return
queue.process(condition, order, action);
    }

        }
    }

    private static PrintQueue<PrintJob> copyQueue(PrintQueue<PrintJob> src) {
        PrintQueue<PrintJob> copy = new PrintQueue<>(src.getPaper());
        src.getAnalyticsStream().forEach(j -> {
            PrintJob c = (j instanceof DoubleSidedJob)
                    ? new DoubleSidedJob(j.getId(), j.getOwnerId(), j.getPages(), j.getCopies(), j.getPriority())
                    : new SingleSidedJob(j.getId(), j.getOwnerId(), j.getPages(), j.getCopies(), j.getPriority());
            copy.add(c);
            if (j.isPrinted()) c.markPrinted();
        });
        return copy;
    }
